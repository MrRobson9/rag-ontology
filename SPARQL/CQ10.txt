A — Evidência (Answer → Citation → RetrievedChunk → Chunk → Document)

PREFIX : <http://purl.org/rag-onto#>

SELECT
  ?answer
  ?cit
  ?rc
  ?chunk
  ?doc
  (STR(?cit)   AS ?cit_str)
  (STR(?rc)    AS ?rc_str)
  (STR(?chunk) AS ?chunk_str)
  (STR(?doc)   AS ?doc_str)
WHERE {
  # (1) Fixa a resposta que queremos explicar
  VALUES ?answer { :answer1 }

  # (2) Resposta -> citações
  ?answer :cites ?cit .

  # (3) Citação -> item recuperado (RetrievedChunk)
  ?cit :refersTo ?rc .

  # (4) Item recuperado -> chunk original
  ?rc :refersTo_1 ?chunk .

  # (5) Documento dono do chunk
  ?doc :hasChunk ?chunk .
}
ORDER BY ?cit ?rc


B — Execuções associadas à resposta (LLMCall e RetrievalExec)

PREFIX : <http://purl.org/rag-onto#>

SELECT
  ?answer
  ?llmCall
  ?retrievalExec
  ?ctx
  (STR(?llmCall)      AS ?llmCall_str)
  (STR(?retrievalExec) AS ?retrievalExec_str)
  (STR(?ctx)          AS ?ctx_str)
WHERE {
  # (1) Fixa a resposta
  VALUES ?answer { :answer1 }

  # (2) A chamada ao LLM cria a resposta
  ?llmCall :creates_2 ?answer .

  # (3) O prompt é entrada da chamada ao LLM
  ?prompt :inputTo ?llmCall .

  # (4) O contexto recuperado foi usado na montagem do prompt
  ?ctx :usedIn ?prompt .

  # (5) A execução de retrieval criou esse contexto
  ?retrievalExec :creates_1 ?ctx .
}


Se você quiser também amarrar o chunk ao IndexingRun (quando existir):

PREFIX : <http://purl.org/rag-onto#>

SELECT DISTINCT
  ?answer
  ?chunk
  ?run
  (STR(?run) AS ?run_str)
WHERE {
  VALUES ?answer { :answer1 }

  # Caminho de evidência até o chunk
  ?answer :cites ?cit .
  ?cit :refersTo ?rc .
  ?rc :refersTo_1 ?chunk .

  # Se houver um IndexingRun que incluiu aquele chunk, retornamos
  OPTIONAL { ?run a :IndexingRun ; :includes_1 ?chunk . }
}
