PREFIX : <http://purl.org/mrrobson9/rag#>

SELECT DISTINCT
  (STR(?doc)    AS ?doc_str)
  (STR(?model)  AS ?embeddingModel_str)
  (STR(?nameV)  AS ?modelName_str)
  (STR(?providerV) AS ?provider_str)
  (STR(?release) AS ?modelRelease_str)
  (STR(?versionV) AS ?version_str)
WHERE {
  VALUES ?doc { :doc1 }

  # Documento -> Chunks
  ?doc :hasChunk ?chunk .

  # EmbeddingExec que processa o chunk
  ?embExec (:processes|:processes_1|:processes_2) ?chunk .

  # EmbeddingExec usa um ModelRelease (o plugin pode gerar uses, uses_1, uses_2...)
  ?embExec (:uses|:uses_1|:uses_2|:uses_3|:uses_4) ?release .

  # O release pertence a um EmbeddingModel
  {
    ?model :hasRelease ?release .
  } UNION {
    ?release :hasRelease ?model .
  } UNION {
    ?model :hasRelease_1 ?release .
  } UNION {
    ?release :hasRelease_1 ?model .
  }

  # Metadados do modelo / release (se existirem)
  OPTIONAL { ?model   :name_2   ?nameV . }
  OPTIONAL { ?model   :provider ?providerV . }
  OPTIONAL { ?release :version  ?versionV . }
}
ORDER BY ?model
