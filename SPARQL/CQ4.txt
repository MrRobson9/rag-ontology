PREFIX : <http://purl.org/mrrobson9/rag#>

SELECT DISTINCT
  (STR(?chunk) AS ?chunk_str)
  (STR(?embExec) AS ?embeddingExec_str)
  (STR(?embConf) AS ?embeddingConf_str)

  (STR(?poolingV) AS ?pooling_str)
  (STR(?normalizeV) AS ?normalize_str)
  (STR(?precisionV) AS ?precision_str)
  (STR(?batchSizeV) AS ?batchSize_str)
WHERE {
  VALUES ?chunk { :chunk1 }

  # Execução de embedding que processa o chunk
  ?embExec (:processes|:processes_1|:processes_2|:processes_3|:processes_4) ?chunk .

  # No seu A-box, EmbeddingExec usa a configuração via :uses_5
  ?embExec :uses_5 ?embConf .
  ?embConf a :EmbeddingConfiguration .

  OPTIONAL { ?embConf :pooling    ?poolingV . }
  OPTIONAL { ?embConf :normalize  ?normalizeV . }
  OPTIONAL { ?embConf :precision  ?precisionV . }
  OPTIONAL { ?embConf :batchSize  ?batchSizeV . }
}
ORDER BY ?embExec
