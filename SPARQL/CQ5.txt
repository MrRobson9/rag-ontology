PREFIX : <http://purl.org/rag-onto#>

SELECT
  ?q
  ?retrievalExec
  ?ctx
  ?rc
  ?chunk
  (STR(?rank)  AS ?rank_str)
  (STR(?score) AS ?score_str)
WHERE {
  # (1) Fixa a query do usuário (exemplo: q1)
  VALUES ?q { :q1 }

  # (2) A query dispara uma execução de recuperação (retrievalExec)
  ?q :triggers ?retrievalExec .

  # (3) A execução cria um "RetrievedContext" (ctx)
  ?retrievalExec :creates_1 ?ctx .

  # (4) O contexto recuperado contém itens (RetrievedChunk / rc)
  ?ctx :hasItem ?rc .

  # (5) Cada item recuperado referencia o Chunk original
  ?rc :refersTo_1 ?chunk .

  # (6) rank/score são opcionais (podem não existir em algumas instâncias)
  OPTIONAL { ?rc :rank  ?rank . }
  OPTIONAL { ?rc :score ?score . }
}
ORDER BY ?rank ?score
