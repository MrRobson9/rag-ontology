PREFIX :     <http://purl.org/mrrobson9/rag#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT
  ?q
  ?retrievalExec
  ?rerankExec
  ?rerankerConf
  (STR(?rerankerLabel)  AS ?reranker_str)
  (STR(?rerankTopKV)    AS ?rerankTopK_str)
  (STR(?rerankFiltersV) AS ?rerankFilters_str)
  ?rc
  (STR(?rankV)          AS ?finalRank_str)
  (STR(?scoreV)         AS ?finalScore_str)
WHERE {
  VALUES ?q { :q1 }

  # Query -> RetrievalExec
  ?q :triggers ?retrievalExec .

  # RetrievalExec -> RerankingExec (existência = houve reranking)
  ?retrievalExec :partOf ?rerankExec .
  ?rerankExec a :RerankingExec .

  # Qual reranker e parâmetros (via config usada pela retrieval exec)
  ?retrievalExec :uses_2 ?rerankerConf .
  OPTIONAL { ?rerankerConf rdfs:label ?rerankerLabel . }
  OPTIONAL { ?rerankerConf :topK      ?rerankTopKV . }
  OPTIONAL { ?rerankerConf :filters_1 ?rerankFiltersV . }

  # Itens retornados com rank/score finais
  ?retrievalExec :produces ?rc .
  OPTIONAL { ?rc :rank  ?rankV . }
  OPTIONAL { ?rc :score ?scoreV . }
}
ORDER BY ?rankV
