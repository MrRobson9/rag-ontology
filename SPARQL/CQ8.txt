PREFIX : <http://purl.org/rag-onto#>

SELECT DISTINCT
  ?answer
  ?doc
  ?chunk
  (STR(?doc)   AS ?doc_str)
  (STR(?chunk) AS ?chunk_str)
WHERE {
  # (1) Resposta gerada com pelo menos uma citação
  ?answer a :GeneratedAnswer ;
          :cites ?cit .

  # (2) A citação aponta para um item recuperado (rc = RetrievedChunk)
  ?cit :refersTo ?rc .

  # (3) O item recuperado referencia o Chunk original
  ?rc  :refersTo_1 ?chunk .

  # (4) O documento ao qual o chunk pertence
  ?doc a :Document ;
       :hasChunk ?chunk .

  # (5) O documento possui algum nível de confidencialidade associado
  ?conf a :ConfidentialLevel ;
        :inheresIn ?doc .
}
ORDER BY ?answer ?doc ?chunk
